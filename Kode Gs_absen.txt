// 1. SETUP DATABASE OTOMATIS
function initSetup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (!ss.getSheetByName("Pegawai")) {
    let sheet = ss.insertSheet("Pegawai");
    sheet.appendRow(["ID_QR", "Nama", "NIP_Gol", "Jabatan", "Status", "Lokasi_Tugas", "Pangkat_Golongan", "Kunci"]);
    sheet.getRange("A1:H1").setFontWeight("bold").setBackground("#003366").setFontColor("#FFFFFF");
  } else {
    let sheet = ss.getSheetByName("Pegawai");
    if (!sheet.getRange("G1").getValue()) {
      sheet.getRange("G1").setValue("Pangkat_Golongan").setFontWeight("bold").setBackground("#003366").setFontColor("#FFFFFF");
    }
    if (!sheet.getRange("H1").getValue()) {
      sheet.getRange("H1").setValue("Kunci").setFontWeight("bold").setBackground("#003366").setFontColor("#FFFFFF");
    }
  }

  if (!ss.getSheetByName("Absensi")) {
    let sheet = ss.insertSheet("Absensi");
    sheet.appendRow(["Timestamp", "Tanggal", "Bulan", "Tahun", "ID_QR", "Nama", "Waktu_Masuk", "Waktu_Pulang", "Jam_Kerja", "Status_Lokasi", "Keterangan", "Alasan", "Lat", "Lng"]);
    sheet.getRange("A1:N1").setFontWeight("bold").setBackground("#003366").setFontColor("#FFFFFF");
  }

  if (!ss.getSheetByName("Kegiatan")) {
    let sheet = ss.insertSheet("Kegiatan");
    sheet.appendRow(["Timestamp", "Tanggal_Kegiatan", "Nama_Kegiatan", "ID_QR", "Nama_Pegawai", "NIP", "Pangkat_Gol", "Jabatan"]);
    sheet.getRange("A1:H1").setFontWeight("bold").setBackground("#10b981").setFontColor("#FFFFFF");
  }

  ensureSheetLibur_();

  if (!ss.getSheetByName("Pengaturan")) {
    let sheet = ss.insertSheet("Pengaturan");
    sheet.appendRow(["Key", "Value"]);
    sheet.getRange("A1:B1").setFontWeight("bold").setBackground("#003366").setFontColor("#FFFFFF");
  }
  ensurePengaturanDefaults_();
}

function ensureSheetLibur_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("Libur") || ss.getSheetByName("libur") || ss.getSheetByName("LIBUR");
  if (!sheet) sheet = ss.insertSheet("Libur");
  const header = ["Tahun", "Bulan", "Tanggal", "Keterangan Libur"];
  const currentHeader = sheet.getRange(1, 1, 1, 4).getDisplayValues()[0].join("").trim();
  if (!currentHeader) sheet.getRange(1, 1, 1, 4).setValues([header]);
  sheet.getRange("A1:D1").setFontWeight("bold").setBackground("#f59e0b").setFontColor("#FFFFFF");
  sheet.setFrozenRows(1);
  sheet.getRange("A:A").setNumberFormat("0");
  sheet.getRange("C:C").setNumberFormat("dd/MM/yyyy");
  try { sheet.autoResizeColumns(1, 4); } catch (e) {}
  return sheet;
}

function triggerDummyDataLibur() {
  const sheet = ensureSheetLibur_();
  const dummy = getDummyDataLibur_();
  const existingRows = sheet.getDataRange().getDisplayValues();
  const existingKeys = new Set();
  for (let i = 1; i < existingRows.length; i++) {
    const r = existingRows[i];
    existingKeys.add(`${String(r[0]).trim()}|${String(r[1]).trim().toUpperCase()}|${String(r[2]).trim()}|${String(r[3]).trim().toUpperCase()}`);
  }
  const rowsToAppend = dummy.filter(r => !existingKeys.has(`${r[0]}|${String(r[1]).toUpperCase()}|${r[2]}|${String(r[3]).toUpperCase()}`));
  if (rowsToAppend.length) {
    sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, 4).setValues(rowsToAppend);
    sheet.getRange(2, 3, sheet.getLastRow() - 1, 1).setNumberFormat("dd/MM/yyyy");
  }
  try { sheet.autoResizeColumns(1, 4); } catch (e) {}
  return { status: true, message: `Dummy Sheet Libur siap. Ditambahkan ${rowsToAppend.length} baris baru. Total data: ${Math.max(0, sheet.getLastRow() - 1)}.` };
}

function buatDummyDataLibur() {
  return triggerDummyDataLibur();
}

// 2. LOGIN & PENGATURAN
function cekLogin(user, pass) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pengaturan");
  const data = sheet.getDataRange().getValues();
  let validUser = "", validPass = "";
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] == "Admin_User") validUser = data[i][1];
    if (data[i][0] == "Admin_Pass") validPass = data[i][1];
  }
  return (user === validUser && pass === validPass) ? { status: true, message: "Login Berhasil!" } : { status: false, message: "Username atau Password salah!" };
}

function getPengaturan() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("Pengaturan");
  if (!sheet) {
    initSetup();
    sheet = ss.getSheetByName("Pengaturan");
    if (!sheet) return {};
  }
  const data = sheet.getDataRange().getValues();
  let setting = {};
  for (let i = 1; i < data.length; i++) setting[data[i][0]] = data[i][1];
  return setting;
}

function simpanPengaturan(dataForm) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pengaturan");
  const data = sheet.getDataRange().getValues();
  let keysInSheet = [];

  for (let i = 1; i < data.length; i++) {
    let key = data[i][0];
    keysInSheet.push(key);
    if (dataForm[key] !== undefined) sheet.getRange(i + 1, 2).setValue(dataForm[key]);
  }
  for (let key in dataForm) {
    if (!keysInSheet.includes(key)) sheet.appendRow([key, dataForm[key]]);
  }
  return { status: true, message: "Seluruh pengaturan & identitas berhasil disimpan!" };
}

// 3. KELOLA PEGAWAI
function tambahPegawai(dataForm) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pegawai");
  const idQR = "PGW-" + new Date().getTime() + Math.floor(Math.random() * 100);
  sheet.appendRow([idQR, dataForm.nama, dataForm.nip, dataForm.jabatan, dataForm.status, dataForm.lokasi, dataForm.golongan, dataForm.kunci || "IYA"]);
  return { status: true, message: "Pegawai berhasil ditambahkan!" };
}

function editPegawai(dataForm) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pegawai");
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === dataForm.idQR) {
      sheet.getRange(i + 1, 2).setValue(dataForm.nama); sheet.getRange(i + 1, 3).setValue(dataForm.nip);
      sheet.getRange(i + 1, 4).setValue(dataForm.jabatan); sheet.getRange(i + 1, 5).setValue(dataForm.status);
      sheet.getRange(i + 1, 6).setValue(dataForm.lokasi); sheet.getRange(i + 1, 7).setValue(dataForm.golongan); sheet.getRange(i + 1, 8).setValue(dataForm.kunci || "IYA");
      return { status: true, message: "Data Pegawai berhasil diperbarui!" };
    }
  }
  return { status: false, message: "Gagal: Pegawai tidak ditemukan!" };
}

function hapusPegawai(idQR) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pegawai");
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === idQR) {
      sheet.deleteRow(i + 1);
      return { status: true, message: "Data Pegawai berhasil dihapus!" };
    }
  }
  return { status: false, message: "Gagal: Pegawai tidak ditemukan!" };
}

function getListPegawai() {
  const data = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Pegawai").getDataRange().getValues();
  data.shift(); return data;
}

// 4. LOGIKA ABSENSI
function formatPesanAbsen_(template, ctx) {
  template = String(template || "");
  const data = ctx || {};
  return template
    .replace(/\{nama\}/g, data.nama || "")
    .replace(/\{waktu\}/g, data.waktu || "")
    .replace(/\{jamKerja\}/g, data.jamKerja || "")
    .replace(/\{lokasi\}/g, data.lokasi || "")
    .replace(/\{tanggal\}/g, data.tanggal || "");
}

function prosesAbsen(qrText, latUser, lngUser, action, gpsAccuracy) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dataPgw = ss.getSheetByName("Pegawai").getDataRange().getValues();
  let pgwFound = null;
  for (let i = 1; i < dataPgw.length; i++) {
    if (dataPgw[i][0] === qrText) { pgwFound = { id: dataPgw[i][0], nama: dataPgw[i][1], lokasiTugas: dataPgw[i][5], kunciJam: String(dataPgw[i][7] || "IYA").toUpperCase() }; break; }
  }
  if (!pgwFound) return { status: 'ERROR', message: "QR Code tidak terdaftar di sistem!" };

  const settings = getPengaturan();
  let targetLat, targetLng, targetRad;
  let lokasiInduk = "Puskesmas";
  let lokasiTag = pgwFound.lokasiTugas || "Puskesmas";
  const lok = String(pgwFound.lokasiTugas || "").toLowerCase();

  if (lok.includes("tebing raya")) {
    lokasiInduk = "Polindes Tebing Raya";
    targetLat = parseFloat(settings.Lat_Pol_TebingRaya); targetLng = parseFloat(settings.Lng_Pol_TebingRaya); targetRad = parseFloat(settings.Rad_Pol_TebingRaya);
  } else if (lok.includes("mungguk bantok")) {
    lokasiInduk = "Polindes Mungguk Bantok";
    targetLat = parseFloat(settings.Lat_Pol_MunggukBantok); targetLng = parseFloat(settings.Lng_Pol_MunggukBantok); targetRad = parseFloat(settings.Rad_Pol_MunggukBantok);
  } else if (lok.includes("kebiau baru")) {
    lokasiInduk = "Polindes Kebiau Baru";
    targetLat = parseFloat(settings.Lat_Pol_KebiauBaru); targetLng = parseFloat(settings.Lng_Pol_KebiauBaru); targetRad = parseFloat(settings.Rad_Pol_KebiauBaru);
  } else if (lok.includes("jerora satu")) {
    lokasiInduk = "Polindes Jerora Satu";
    targetLat = parseFloat(settings.Lat_Pol_JeroraSatu); targetLng = parseFloat(settings.Lng_Pol_JeroraSatu); targetRad = parseFloat(settings.Rad_Pol_JeroraSatu);
  } else {
    lokasiInduk = "Puskesmas";
    targetLat = parseFloat(settings.Lat_Puskesmas); targetLng = parseFloat(settings.Lng_Puskesmas); targetRad = parseFloat(settings.Rad_Puskesmas);
  }

  const distance = hitungJarak(latUser, lngUser, targetLat, targetLng);
  const accuracy = Math.max(0, parseFloat(gpsAccuracy) || 0);
  const accuracyTolerance = Math.min(accuracy, Math.max(50, targetRad));
  const effectiveDistance = Math.max(0, distance - accuracyTolerance);
  const localTime = new Date();
  const tglStr = Utilities.formatDate(localTime, "Asia/Jakarta", "dd/MM/yyyy");
  const wktStr = Utilities.formatDate(localTime, "Asia/Jakarta", "HH:mm:ss");
  const nowMenit = waktuKeMenit(wktStr);

  let outOfRadius = effectiveDistance > targetRad;
  let lokasiStr = `Jarak: ${Math.round(distance)}m (akurasi ±${Math.round(accuracy)}m, toleransi ${Math.round(accuracyTolerance)}m, efektif ${Math.round(effectiveDistance)}m)`;
  if (lokasiInduk !== "Puskesmas") { lokasiStr = outOfRadius ? `LUAR AREA ${lokasiInduk.toUpperCase()} (${lokasiStr})` : `${lokasiInduk.toUpperCase()} (${lokasiStr})`; }
  else { lokasiStr = outOfRadius ? `LUAR AREA (${lokasiStr})` : `PUSKESMAS (${lokasiStr})`; }

  const sheetAbsen = ss.getSheetByName("Absensi");
  const dataAbsen = sheetAbsen.getDataRange().getValues();
  let rowIndex = -1; let existingData = null;

  for (let i = 1; i < dataAbsen.length; i++) {
    let absenTgl = dataAbsen[i][1];
    if (absenTgl instanceof Date) absenTgl = Utilities.formatDate(absenTgl, "Asia/Jakarta", "dd/MM/yyyy");
    else absenTgl = String(absenTgl).replace(/'/g, "");

    if (dataAbsen[i][4] === qrText && absenTgl === tglStr) { rowIndex = i + 1; existingData = dataAbsen[i]; break; }
  }

  const openPrev = cariAbsenBelumPulangSebelumnya_(dataAbsen, qrText, tglStr);
  if (openPrev) {
    return {
      status: 'NEED_PREVIOUS_PULANG',
      idQR: pgwFound.id,
      nama: pgwFound.nama,
      tanggal: openPrev.tanggal,
      waktuMasuk: openPrev.waktuMasuk,
      lokasiInduk: lokasiInduk,
      lokasiTag: lokasiTag,
      message: `Masih ada absen hari sebelumnya (${openPrev.tanggal}) yang belum memiliki jam pulang.`
    };
  }

  if (action === "MASUK") {
    const jamMasukMulai = settings.Jam_Masuk_Mulai || "00:00";
    const jamMasukAkhir = settings.Jam_Masuk_Akhir || "23:59";
    const jamMasukMulaiTxt = formatJamSetting(jamMasukMulai);
    const jamMasukAkhirTxt = formatJamSetting(jamMasukAkhir);
    if (pgwFound.kunciJam !== "TIDAK" && !isDalamRentangJam(nowMenit, jamMasukMulai, jamMasukAkhir)) {
      return { status: 'ERROR', lokasiInduk: lokasiInduk, lokasiTag: lokasiTag, message: `Absen MASUK hanya diizinkan jam ${jamMasukMulaiTxt} - ${jamMasukAkhirTxt}.` };
    }
    if (rowIndex > -1) {
      let ket = String(existingData[10]).toUpperCase();
      let wktMsk = existingData[6];
      if (wktMsk instanceof Date) wktMsk = Utilities.formatDate(wktMsk, "Asia/Jakarta", "HH:mm:ss");
      return { status: 'ERROR', message: `Anda sudah tercatat ${ket} hari ini pada pukul ${wktMsk}.` };
    }
    if (outOfRadius) {
      return { status: 'ERROR', idQR: pgwFound.id, nama: pgwFound.nama, lokasiInduk: lokasiInduk, lokasiTag: lokasiTag, message: `Absen MASUK ditolak. Anda berada di luar lokasi resmi Puskesmas/Polindes (${Math.round(effectiveDistance)}m efektif, akurasi ±${Math.round(accuracy)}m). Gunakan tombol scan IZIN/SAKIT/TL jika memang tidak hadir di lokasi kerja.` };
    }
    const blnStr = Utilities.formatDate(localTime, "Asia/Jakarta", "MMMM"); const thnStr = Utilities.formatDate(localTime, "Asia/Jakarta", "yyyy");
    sheetAbsen.appendRow([localTime, tglStr, blnStr, thnStr, pgwFound.id, pgwFound.nama, wktStr, "-", "-", lokasiStr, "HADIR", "-", latUser, lngUser]);
    const pesanMasuk = formatPesanAbsen_(settings.Pesan_Masuk_Berhasil || "Absen MASUK berhasil. Selamat bekerja, {nama}!", { nama: pgwFound.nama, waktu: wktStr, lokasi: lokasiInduk, tanggal: tglStr });
    return { status: 'SUCCESS_MASUK', waktu: wktStr, lokasiInduk: lokasiInduk, lokasiTag: lokasiTag, message: pesanMasuk };
  }
  else if (action === "PULANG") {
    const jamPulangMulai = settings.Jam_Pulang_Mulai || "00:00";
    const jamPulangAkhir = settings.Jam_Pulang_Akhir || "23:59";
    const jamPulangMulaiTxt = formatJamSetting(jamPulangMulai);
    const jamPulangAkhirTxt = formatJamSetting(jamPulangAkhir);
    if (pgwFound.kunciJam !== "TIDAK" && !isDalamRentangJam(nowMenit, jamPulangMulai, jamPulangAkhir)) {
      return { status: 'ERROR', lokasiInduk: lokasiInduk, lokasiTag: lokasiTag, message: `Absen PULANG hanya diizinkan jam ${jamPulangMulaiTxt} - ${jamPulangAkhirTxt}.` };
    }
    if (rowIndex === -1) return { status: 'ERROR', message: `Anda belum melakukan Absen MASUK hari ini!` };

    let ket = String(existingData[10]).toUpperCase();
    if (ket !== "HADIR") return { status: 'ERROR', message: `Status hari ini adalah ${ket}. Tidak perlu absen pulang.` };

    let waktuPulang = existingData[7];
    if (waktuPulang !== "-" && waktuPulang !== "") {
      if (waktuPulang instanceof Date) waktuPulang = Utilities.formatDate(waktuPulang, "Asia/Jakarta", "HH:mm:ss");
      return { status: 'ERROR', message: `Anda sudah melakukan Absen PULANG hari ini pada pukul ${waktuPulang}.` };
    }

    if (outOfRadius) return { status: 'ERROR', message: `Absen PULANG ditolak. Anda berada di luar radius penugasan (${Math.round(effectiveDistance)}m efektif, akurasi ±${Math.round(accuracy)}m).` };

    let wktMasukRaw = existingData[6];
    let wktMasukStr = wktMasukRaw instanceof Date ? Utilities.formatDate(wktMasukRaw, "Asia/Jakarta", "HH:mm:ss") : String(wktMasukRaw);
    let jamKerja = hitungJamKerja(wktMasukStr, wktStr);

    sheetAbsen.getRange(rowIndex, 8).setValue(wktStr); sheetAbsen.getRange(rowIndex, 9).setValue(jamKerja); sheetAbsen.getRange(rowIndex, 10).setValue(lokasiStr);
    const pesanPulang = formatPesanAbsen_(settings.Pesan_Pulang_Berhasil || "Absen PULANG berhasil. Terima kasih, {nama}.", { nama: pgwFound.nama, waktu: wktStr, jamKerja: jamKerja, lokasi: lokasiInduk, tanggal: tglStr });
    return { status: 'SUCCESS_PULANG', waktu: wktStr, jamKerja: jamKerja, lokasiInduk: lokasiInduk, lokasiTag: lokasiTag, message: pesanPulang };
  }
}

function cariAbsenBelumPulangSebelumnya_(dataAbsen, idQR, tglHariIni) {
  const today = parseTanggalAbsensi(tglHariIni);
  let latest = null;

  for (let i = 1; i < dataAbsen.length; i++) {
    if (dataAbsen[i][4] !== idQR) continue;

    const tgl = parseTanggalAbsensi(dataAbsen[i][1]);
    if (!tgl || !today || tgl.getTime() >= today.getTime()) continue;

    const ket = String(dataAbsen[i][10] || "").toUpperCase().trim();
    const masukRaw = dataAbsen[i][6];

function waktuKeMenit(hhmmss) {
  if (hhmmss instanceof Date) {
    return hhmmss.getHours() * 60 + hhmmss.getMinutes();
  }
  if (typeof hhmmss === "number" && !isNaN(hhmmss)) {
    let totalMenit = Math.round((hhmmss % 1) * 24 * 60);
    return ((totalMenit % 1440) + 1440) % 1440;
  }

  let clean = String(hhmmss || "00:00").trim().replace(/'/g, "");
  let m = clean.match(/(\d{1,2}):(\d{2})/);
  if (!m) return 0;
  let jam = parseInt(m[1], 10) || 0;
  let menit = parseInt(m[2], 10) || 0;
  return (jam * 60 + menit) % 1440;
}

function isDalamRentangJam(nowMenit, mulaiHHMM, akhirHHMM) {
  const start = waktuKeMenit(mulaiHHMM);
  const end = waktuKeMenit(akhirHHMM);
  if (start <= end) return nowMenit >= start && nowMenit <= end;
  return nowMenit >= start || nowMenit <= end;
}

function formatJamSetting(v) {
  if (v instanceof Date) return Utilities.formatDate(v, "Asia/Jakarta", "HH:mm");
  if (typeof v === "number" && !isNaN(v)) {
    let menit = waktuKeMenit(v);
    let h = String(Math.floor(menit / 60)).padStart(2, "0");
    let m = String(menit % 60).padStart(2, "0");
    return `${h}:${m}`;
  }
  let s = String(v || "").trim();
  let match = s.match(/(\d{1,2}):(\d{2})/);
  if (!match) return "00:00";
  return `${String(parseInt(match[1], 10)).padStart(2, "0")}:${match[2]}`;
}

function hitungJamKerja(masukStr, pulangStr) {
  try {
    masukStr = masukStr.replace(/'/g, "").trim(); pulangStr = pulangStr.replace(/'/g, "").trim();
    let m = masukStr.split(":"); let p = pulangStr.split(":");
    let dateM = new Date(2000, 0, 1, m[0], m[1], m[2] || 0); let dateP = new Date(2000, 0, 1, p[0], p[1], p[2] || 0);
    let diffMs = dateP - dateM;
    if(diffMs < 0) return "0 Jam 0 Menit";
    let diffHrs = Math.floor((diffMs % 86400000) / 3600000); let diffMins = Math.round(((diffMs % 86400000) % 3600000) / 60000);
    return `${diffHrs} Jam ${diffMins} Menit`;
  } catch(e) { return "-"; }
}

function hitungJarak(lat1, lon1, lat2, lon2) {
  var R = 6371e3; var p1 = lat1 * Math.PI/180; var p2 = lat2 * Math.PI/180;
  var dp = (lat2-lat1) * Math.PI/180; var dl = (lon2-lon1) * Math.PI/180;
  var a = Math.sin(dp/2) * Math.sin(dp/2) + Math.cos(p1) * Math.cos(p2) * Math.sin(dl/2) * Math.sin(dl/2);
  var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
  return R * c;
}

function parseTanggalAbsensi(value) {
  if (value instanceof Date) return new Date(value.getFullYear(), value.getMonth(), value.getDate());
  let s = String(value || "").replace(/'/g, "").trim();
  let m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (!m) return null;
  return new Date(parseInt(m[3], 10), parseInt(m[2], 10) - 1, parseInt(m[1], 10));
}

function getDaftarCutiBulanIni() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dataAbsen = ss.getSheetByName("Absensi").getDataRange().getValues();
  const dataPegawai = ss.getSheetByName("Pegawai").getDataRange().getValues();

  const now = new Date();
  const bulanNow = now.getMonth();
  const tahunNow = now.getFullYear();
  const bulanLabel = Utilities.formatDate(now, "Asia/Jakarta", "MMMM");

  let pegawaiMap = {};
  for (let i = 1; i < dataPegawai.length; i++) {
    pegawaiMap[dataPegawai[i][0]] = { nama: dataPegawai[i][1] || "-", jabatan: dataPegawai[i][3] || "-" };
  }

  let cutiMap = {};
  for (let i = 1; i < dataAbsen.length; i++) {
    let ket = String(dataAbsen[i][10] || "").toUpperCase().trim();
    if (ket !== "CUTI") continue;

    let d = parseTanggalAbsensi(dataAbsen[i][1]);
    if (!d || d.getMonth() !== bulanNow || d.getFullYear() !== tahunNow) continue;

    let idQR = dataAbsen[i][4];
    if (!cutiMap[idQR]) cutiMap[idQR] = { tanggal: [], nama: pegawaiMap[idQR] ? pegawaiMap[idQR].nama : (dataAbsen[i][5] || "-"), jabatan: pegawaiMap[idQR] ? pegawaiMap[idQR].jabatan : "-" };
    cutiMap[idQR].tanggal.push(new Date(d));
  }

  let hasil = [];
  Object.keys(cutiMap).forEach(idQR => {
    let dates = cutiMap[idQR].tanggal
      .sort((a, b) => a - b)
      .map(d => Utilities.formatDate(d, "Asia/Jakarta", "dd/MM/yyyy"));

    let uniqueDates = Array.from(new Set(dates));
    let periods = [];
    let start = null, prev = null;
    uniqueDates.forEach((ds, idx) => {
      let parts = ds.split('/');
      let current = new Date(parseInt(parts[2], 10), parseInt(parts[1], 10) - 1, parseInt(parts[0], 10));
      if (!start) { start = current; prev = current; }
      else {
        let nextExpected = new Date(prev);
        nextExpected.setDate(nextExpected.getDate() + 1);
        if (current.getTime() !== nextExpected.getTime()) {
          periods.push(`${Utilities.formatDate(start, "Asia/Jakarta", "dd/MM")} - ${Utilities.formatDate(prev, "Asia/Jakarta", "dd/MM")}`);
          start = current;
        }
        prev = current;
      }
      if (idx === uniqueDates.length - 1) {
        periods.push(`${Utilities.formatDate(start, "Asia/Jakarta", "dd/MM")} - ${Utilities.formatDate(prev, "Asia/Jakarta", "dd/MM")}`);
      }
    });

    hasil.push({
      id_qr: idQR,
      nama: cutiMap[idQR].nama,
      jabatan: cutiMap[idQR].jabatan,
      total_hari: uniqueDates.length,
      periode: periods.join(", ")
    });
  });

  hasil.sort((a, b) => b.total_hari - a.total_hari || a.nama.localeCompare(b.nama));

  return {
    bulan: bulanLabel,
    tahun: tahunNow,
    data: hasil
  };
}

function getRataJamHariIni() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Absensi");
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  const tglStr = Utilities.formatDate(new Date(), "Asia/Jakarta", "dd/MM/yyyy");
  let totalMasuk = 0, countMasuk = 0, totalPulang = 0, countPulang = 0;

  rows.forEach(r => {
    let tgl = r[1];
    if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, "Asia/Jakarta", "dd/MM/yyyy");
    else tgl = String(tgl).replace(/'/g, "");
    if (tgl !== tglStr) return;
    const status = String(r[10] || "").toUpperCase();
    if (status !== "HADIR") return;
    const masuk = waktuToDetik_(r[6]);
    const pulang = waktuToDetik_(r[7]);
    if (masuk !== null) { totalMasuk += masuk; countMasuk++; }
    if (pulang !== null) { totalPulang += pulang; countPulang++; }
  });

  return {
    rataMasuk: countMasuk ? detikToJam_(Math.round(totalMasuk / countMasuk)) : "--:--",
    rataPulang: countPulang ? detikToJam_(Math.round(totalPulang / countPulang)) : "--:--",
    totalMasuk: countMasuk,
    totalPulang: countPulang
  };
}

function waktuToDetik_(value) {
  if (!value || value === "-") return null;
  let txt = value instanceof Date ? Utilities.formatDate(value, "Asia/Jakarta", "HH:mm:ss") : String(value).replace(/'/g, "");
  const m = txt.match(/^(\d{1,2}):(\d{2})(?::(\d{2}))?/);
  if (!m) return null;
  return (parseInt(m[1], 10) * 3600) + (parseInt(m[2], 10) * 60) + (parseInt(m[3] || "0", 10));
}

function detikToJam_(total) {
  const h = Math.floor(total / 3600) % 24;
  const m = Math.floor((total % 3600) / 60);
  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
}

    if (tanggalAbsenSheet instanceof Date) tanggalAbsenSheet = Utilities.formatDate(tanggalAbsenSheet, "Asia/Jakarta", "dd/MM/yyyy");
    else tanggalAbsenSheet = String(tanggalAbsenSheet).replace(/'/g, "");
    if (tanggalAbsenSheet === formattedDate && a[4]) absenById[String(a[4])] = a;
  }

  let hasilHarian = [];

  dPgw.forEach(p => {
    let statusHarian = "BELUM ABSEN";
    let wktMasuk = "-", wktPulang = "-", jamKerja = "-", alasan = "-", jarakInfo = "-";
    const a = absenById[String(p[0])] || null;

    if (a) {
      let rawStatus = String(a[10]).toUpperCase().trim();
      let mRaw = a[6];
      let pRaw = a[7];
      wktMasuk = mRaw instanceof Date ? Utilities.formatDate(mRaw, "Asia/Jakarta", "HH:mm:ss") : String(mRaw).replace(/'/g, "");
      wktPulang = pRaw instanceof Date ? Utilities.formatDate(pRaw, "Asia/Jakarta", "HH:mm:ss") : String(pRaw).replace(/'/g, "");
      jamKerja = a[8] || "-";
      alasan = a[11] || "-";
      jarakInfo = a[9] || "-";

      if (rawStatus === "HADIR" && wktMasuk !== "-" && wktMasuk !== "" &&
          (wktPulang === "-" || wktPulang === "" || !wktPulang)) {
        statusHarian = "BELUM ABSEN PULANG";
      } else {
        statusHarian = rawStatus;
      }
    }

    hasilHarian.push({
      id_qr: p[0], nama: p[1], jabatan: p[3],
      wktMasuk: wktMasuk, wktPulang: wktPulang, jamKerja: jamKerja,
      keterangan: statusHarian, alasan: alasan, jarakInfo: jarakInfo
    });
  });

  return {
    data: hasilHarian,
    namaPuskesmas: settings.Nama_Puskesmas || "PUSKESMAS",
    kapusNama: settings.Kepala_Nama || "",
    kapusNip: settings.Kepala_NIP || ""
  };
}

// 6. REKAPITULASI PDF BULANAN
function getRekapitulasi(bulan, tahun, hariKerjaInput) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dPgw = ss.getSheetByName("Pegawai").getDataRange().getValues(); dPgw.shift();
  const dAbsen = ss.getSheetByName("Absensi").getDataRange().getValues(); dAbsen.shift();
  const settings = getPengaturan();

  const targetHari = parseInt(hariKerjaInput) || 17;
  let rekap = [];

  dPgw.forEach(p => {
    let hadir = 0, sakit = 0, izin = 0, cuti = 0, tl = 0, alpaTercatat = 0;
    let totalMenitBulanIni = 0;

    dAbsen.forEach(a => {
      let isMatchMonth = false;
      let tanggalRaw = a[1];
      if (tanggalRaw instanceof Date) {
        isMatchMonth = (tanggalRaw.getMonth() + 1 === parseInt(getMonthNumber_(bulan), 10) && tanggalRaw.getFullYear() === parseInt(tahun, 10));
      } else {
        let tglTxt = String(tanggalRaw || "").replace(/'/g, "").trim();
        let parts = tglTxt.split("/");
        if (parts.length === 3) {
          isMatchMonth = (parseInt(parts[1], 10) === parseInt(getMonthNumber_(bulan), 10) && parseInt(parts[2], 10) === parseInt(tahun, 10));
        }
      }

      if(a[4] === p[0] && isMatchMonth) {
        let ket = String(a[10]).toUpperCase().trim();
        if(ket === "HADIR") hadir++; else if(ket === "SAKIT") sakit++; else if(ket === "IZIN") izin++; else if(ket === "CUTI") cuti++; else if(ket === "TL" || ket === "TUGAS LUAR") tl++; else if(ket === "ALPA") alpaTercatat++;
        if(a[8] && a[8] !== "-") {
            let jkStr = String(a[8]); let matchJam = jkStr.match(/(\d+)\s*Jam/i); let matchMenit = jkStr.match(/(\d+)\s*Menit/i);
            if(matchJam) totalMenitBulanIni += parseInt(matchJam[1]) * 60;
            if(matchMenit) totalMenitBulanIni += parseInt(matchMenit[1]);
        }
      }
    });

    let totalMasuk = hadir + sakit + izin + cuti + tl + alpaTercatat;
    let alpaSisa = (targetHari - totalMasuk) > 0 ? (targetHari - totalMasuk) : 0;
    let totalAlpa = alpaTercatat + alpaSisa;
    let totalJamFormat = Math.floor(totalMenitBulanIni / 60) + " Jam " + (totalMenitBulanIni % 60) + " Menit";
    let strNipGol = String(p[2] || ""); let extractedGol = p[6] || "-";
    if (extractedGol === "-" && strNipGol.includes("/")) { extractedGol = strNipGol.split("/").pop().replace(")","").trim(); }

    rekap.push({
      nama_nip: p[1] + (strNipGol && strNipGol !== '-' ? "\n" + strNipGol : ""),
      gol: extractedGol, jabatan: p[3], status: p[4] + (p[5].includes("Polindes") ? "\n(Polindes)" : ""),
      hari_kerja: targetHari, hadir: hadir, sakit: sakit, izin: izin, cuti: cuti, tl: tl, alpa: totalAlpa, total_jam: totalJamFormat
    });
  });

  return {
    rekapData: rekap,
    namaPuskesmas: settings.Nama_Puskesmas || "UPTD PUSKESMAS TANJUNG PURI SINTANG",
    kapusNama: settings.Kepala_Nama || ".......................", kapusNip: settings.Kepala_NIP || "NIP. ....................."
  };
}

function getMonthNumber_(bulanName) {
  const map = {
    "JANUARI": "01", "FEBRUARI": "02", "MARET": "03", "APRIL": "04",
    "MEI": "05", "JUNI": "06", "JULI": "07", "AGUSTUS": "08",
    "SEPTEMBER": "09", "OKTOBER": "10", "NOVEMBER": "11", "DESEMBER": "12"
  };
  return map[String(bulanName || "").toUpperCase()] || "01";
}

function getKalenderKerja(bulanName, tahun) {
  const monthMap = { "JANUARI":0,"FEBRUARI":1,"MARET":2,"APRIL":3,"MEI":4,"JUNI":5,"JULI":6,"AGUSTUS":7,"SEPTEMBER":8,"OKTOBER":9,"NOVEMBER":10,"DESEMBER":11 };
  const monthName = String(bulanName || "").toUpperCase().replace(/[^A-Z]/g, "").trim();
  const m = monthMap[monthName];
  const y = parseInt(tahun, 10);
  if (m === undefined || isNaN(y)) return { targetHariKerja: 0, detailLibur: [], sumber: "Database Libur" };

  const start = new Date(y, m, 1, 0, 0, 0);
  const end = new Date(y, m + 1, 1, 0, 0, 0);
  const monthNumber = m + 1;

  let weekdays = 0;
  for (let d = new Date(start); d < end; d.setDate(d.getDate() + 1)) {
    const day = d.getDay();
    if (day !== 0 && day !== 6) weekdays++;
  }

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const allSheets = ss.getSheets();
    let sheet = ss.getSheetByName("Libur") || allSheets.find(sh => String(sh.getName()).toLowerCase().trim() === "libur");
    if (!sheet) {
      return {
        targetHariKerja: weekdays,
        totalWeekdays: weekdays,
        totalLiburNasionalWeekday: 0,
        detailLibur: [],
        sumber: "Database Libur belum ditemukan",
        rekomendasiAI: "Database Libur belum ditemukan. Pastikan nama sheet persis Libur dengan kolom Tahun, Bulan, Tanggal, Keterangan Libur."
      };
    }

    const lastRow = sheet.getLastRow();
    const lastCol = Math.max(sheet.getLastColumn(), 4);
    if (lastRow < 2) {
      return {
        targetHariKerja: weekdays,
        totalWeekdays: weekdays,
        totalLiburNasionalWeekday: 0,
        detailLibur: [],
        sumber: "Database Libur kosong",
        rekomendasiAI: "Database Libur masih kosong. Isi kolom Tahun, Bulan, Tanggal, Keterangan Libur untuk menghitung otomatis."
      };
    }

    const values = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
    const header = values[0].map(h => String(h || "").toLowerCase().replace(/[^a-z]/g, "").trim());
    const findCol = (names, fallback) => {
      for (let i = 0; i < header.length; i++) if (names.includes(header[i])) return i;
      return fallback;
    };
    const idxTahun = findCol(["tahun", "year"], 0);
    const idxBulan = findCol(["bulan", "month"], 1);
    const idxTanggal = findCol(["tanggal", "tgl", "date"], 2);
    const idxKet = findCol(["keteranganlibur", "keterangan", "ket", "nama", "namalibur"], 3);

    const holidayMap = {};
    const toDateStr = (val, fallbackYear) => {
      if (!val) return "";
      const raw = String(val).trim();
      let m1 = raw.match(/^(\d{4})[\/\-.](\d{1,2})[\/\-.](\d{1,2})/);
      if (m1) return `${m1[1]}-${String(m1[2]).padStart(2, "0")}-${String(m1[3]).padStart(2, "0")}`;
      let m2 = raw.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})$/);
      if (m2) return `${m2[3]}-${String(m2[2]).padStart(2, "0")}-${String(m2[1]).padStart(2, "0")}`;
      let m3 = raw.match(/^(\d{1,2})[\/\-.](\d{1,2})$/);
      if (m3 && fallbackYear) return `${fallbackYear}-${String(m3[2]).padStart(2, "0")}-${String(m3[1]).padStart(2, "0")}`;
      return "";
    };
    const monthToNumber = (value) => {
      const txt = String(value || "").toUpperCase().replace(/[^A-Z0-9]/g, "").trim();
      if (monthMap[txt] !== undefined) return monthMap[txt] + 1;
      const n = parseInt(txt, 10);
      return isNaN(n) ? 0 : n;
    };

    let parsedRows = 0;
    for (let i = 1; i < values.length; i++) {
      const row = values[i] || [];
      const rowYearRaw = String(row[idxTahun] || "").trim();
      const rowYear = parseInt(rowYearRaw.replace(/[^0-9]/g, ""), 10);
      const rowMonthNumber = monthToNumber(row[idxBulan]);
      const dateStr = toDateStr(row[idxTanggal], rowYear || y);
      const dateParts = dateStr.split("-");
      const dateYear = parseInt(dateParts[0], 10);
      const dateMonth = parseInt(dateParts[1], 10);
      const validDate = dateParts.length === 3 && !isNaN(dateYear) && !isNaN(dateMonth);
      if (!validDate) continue;
      parsedRows++;
      if (dateYear !== y || dateMonth !== monthNumber) continue;
      // Jika kolom Tahun/Bulan terisi, jadikan penguat; kalau kosong, tanggal tetap menjadi sumber utama.
      if (!isNaN(rowYear) && rowYear !== y) continue;
      if (rowMonthNumber && rowMonthNumber !== monthNumber) continue;
      holidayMap[dateStr] = { tgl: dateStr, nama: String(row[idxKet] || "Libur Nasional").trim() || "Libur Nasional" };
    }

    const detailLibur = Object.values(holidayMap).sort((a,b)=>a.tgl.localeCompare(b.tgl));
    const holidayWeekdays = detailLibur.filter(h => {
      const d = new Date(h.tgl + "T00:00:00");
      const day = d.getDay();
      return day !== 0 && day !== 6;
    });

    return {
      targetHariKerja: Math.max(0, weekdays - holidayWeekdays.length),
      totalWeekdays: weekdays,
      totalLiburNasionalWeekday: holidayWeekdays.length,
      detailLibur: holidayWeekdays,
      sumber: "Database Libur",
      rekomendasiAI: `Data dibaca dari Database Libur: ${holidayWeekdays.length} libur pada hari kerja untuk bulan ${bulanName} ${tahun}. Baris tanggal valid terbaca: ${parsedRows}. Target hari kerja awal: ${Math.max(0, weekdays - holidayWeekdays.length)}.`
    };
  } catch (err) {
    return {
      targetHariKerja: weekdays,
      totalWeekdays: weekdays,
      totalLiburNasionalWeekday: 0,
      detailLibur: [],
      sumber: "Database Libur error",
      rekomendasiAI: `Database Libur gagal dibaca: ${err && err.message ? err.message : err}. Cek format kolom Tahun, Bulan, Tanggal, Keterangan Libur.`
    };
  }
}

// 7. ABSENSI MANUAL KEGIATAN
function getPegawaiEligibleKegiatan(tglStr) {
  let parts = tglStr.split('-');
  let formattedDate = `${parts[2]}/${parts[1]}/${parts[0]}`;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dPgw = ss.getSheetByName("Pegawai").getDataRange().getValues(); dPgw.shift();
  const dAbsen = ss.getSheetByName("Absensi").getDataRange().getValues(); dAbsen.shift();
  const settings = getPengaturan();

  const statusById = {};
  for (let i = 0; i < dAbsen.length; i++) {
    const a = dAbsen[i];
    let tanggalAbsenSheet = a[1];
    if (tanggalAbsenSheet instanceof Date) tanggalAbsenSheet = Utilities.formatDate(tanggalAbsenSheet, "Asia/Jakarta", "dd/MM/yyyy");
    else tanggalAbsenSheet = String(tanggalAbsenSheet).replace(/'/g, "");
    if (tanggalAbsenSheet === formattedDate && a[4]) statusById[String(a[4])] = String(a[10] || "BELUM ABSEN").toUpperCase();
  }

  let eligible = [];
  let counts = { tersedia: 0, sakit: 0, izin: 0, tl: 0, cuti: 0 };

  dPgw.forEach(p => {
    let statusAbsen = statusById[String(p[0])] || "BELUM ABSEN";
    if (statusAbsen === "SAKIT") counts.sakit++;
    else if (statusAbsen === "IZIN") counts.izin++;
    else if (statusAbsen === "TL" || statusAbsen === "TUGAS LUAR") counts.tl++;
    else if (statusAbsen === "CUTI") counts.cuti++;
    else {
      counts.tersedia++;
      eligible.push({ id_qr: p[0], nama: p[1], nip: p[2] && p[2] !== '-' ? p[2] : "", jabatan: p[3], golongan: p[6] || "-" });
    }
  });

  return { eligibleData: eligible, counts: counts, namaPuskesmas: settings.Nama_Puskesmas || "PUSKESMAS", kapusNama: settings.Kepala_Nama || "", kapusNip: settings.Kepala_NIP || "" };
}
