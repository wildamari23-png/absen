        function updateDistanceInfo() {
            const text = document.getElementById('distanceText');
            const badge = document.getElementById('distanceBadge');
            const list = document.getElementById('distanceList');
            if (!text || !badge || !list || !currentUserLat || !currentUserLng) return;
            if (!isManualCheckingLocation) return;

            const lokasiRefs = [
                { nama: "Puskesmas", lat: parseFloat(appSettingsCache.Lat_Puskesmas || 0), lng: parseFloat(appSettingsCache.Lng_Puskesmas || 0), rad: parseFloat(appSettingsCache.Rad_Puskesmas || 100) },
                { nama: "Polindes Tebing Raya", lat: parseFloat(appSettingsCache.Lat_Pol_TebingRaya || 0), lng: parseFloat(appSettingsCache.Lng_Pol_TebingRaya || 0), rad: parseFloat(appSettingsCache.Rad_Pol_TebingRaya || 100) },
                { nama: "Polindes Mungguk Bantok", lat: parseFloat(appSettingsCache.Lat_Pol_MunggukBantok || 0), lng: parseFloat(appSettingsCache.Lng_Pol_MunggukBantok || 0), rad: parseFloat(appSettingsCache.Rad_Pol_MunggukBantok || 100) },
                { nama: "Polindes Kebiau Baru", lat: parseFloat(appSettingsCache.Lat_Pol_KebiauBaru || 0), lng: parseFloat(appSettingsCache.Lng_Pol_KebiauBaru || 0), rad: parseFloat(appSettingsCache.Rad_Pol_KebiauBaru || 100) },
                { nama: "Polindes Jerora Satu", lat: parseFloat(appSettingsCache.Lat_Pol_JeroraSatu || 0), lng: parseFloat(appSettingsCache.Lng_Pol_JeroraSatu || 0), rad: parseFloat(appSettingsCache.Rad_Pol_JeroraSatu || 100) }
            ];

            let terdekat = { nama: '-', jarak: Number.POSITIVE_INFINITY, rad: 100 };
            const htmlRows = lokasiRefs.map(l => {
                if (!l.lat || !l.lng) return `<div class="px-2 py-1 rounded-lg bg-gray-100 text-gray-500"><b>${l.nama}:</b> Koordinat belum diset</div>`;
                const d = hitungJarakClient(currentUserLat, currentUserLng, l.lat, l.lng);
                if (d < terdekat.jarak) terdekat = { nama: l.nama, jarak: d, rad: l.rad };
                let cls = d <= l.rad ? 'bg-green-50 text-green-700 border border-green-100' : 'bg-amber-50 text-amber-700 border border-amber-100';
                return `<div class="px-2 py-1 rounded-lg ${cls}"><b>${l.nama}:</b> ${Math.round(d)} m <span class="opacity-70">(R ${Math.round(l.rad)}m)</span></div>`;
            }).join('');
            list.innerHTML = htmlRows;
            list.classList.remove('hidden');
            const loading = document.getElementById('distanceLoading');
            if (loading) { loading.classList.add('hidden'); loading.classList.remove('flex'); }
            const btn = document.getElementById('btnCekLokasi');
            if (btn) { btn.disabled = false; btn.innerHTML = '<i class="fa-solid fa-rotate-right mr-1"></i>Cek Ulang Lokasi'; }

            text.innerHTML = `Terdekat saat ini: <b>${terdekat.nama}</b> (${Math.round(terdekat.jarak)} meter).`;
            if (terdekat.jarak <= terdekat.rad) {
                badge.innerText = "Siap Absen";
                badge.className = "text-[10px] font-extrabold px-2 py-1 rounded-full bg-green-100 text-green-700";
            } else if (terdekat.jarak <= terdekat.rad * 2) {
                badge.innerText = "Perlu Mendekat";
                badge.className = "text-[10px] font-extrabold px-2 py-1 rounded-full bg-amber-100 text-amber-700";
            } else {
                badge.innerText = "Terlalu Jauh";
                badge.className = "text-[10px] font-extrabold px-2 py-1 rounded-full bg-red-100 text-red-700";
            }
        }

        function hitungJarakClient(lat1, lon1, lat2, lon2) {
            const R = 6371e3;
            const p1 = lat1 * Math.PI / 180, p2 = lat2 * Math.PI / 180;
            const dp = (lat2 - lat1) * Math.PI / 180, dl = (lon2 - lon1) * Math.PI / 180;
            const a = Math.sin(dp / 2) ** 2 + Math.cos(p1) * Math.cos(p2) * Math.sin(dl / 2) ** 2;
            return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)));
        }

        function handleScanResult(decodedText, actionType, onFinish = null) {
            let boundQr = localStorage.getItem('pusk_sintang_bound_qr');
            if (boundQr && boundQr !== decodedText) {
                Swal.fire({ icon: 'error', title: 'Titip Absen Ditolak! 🚫', text: 'HP ini sudah dikunci untuk akun pegawai lain. Anda harus menggunakan HP Anda sendiri.', confirmButtonColor: '#d33' });
                if (onFinish) onFinish();
                return;
            }
            if (!boundQr) {
                Swal.fire({
                    title: 'Kunci Perangkat', html: '<p class="text-sm">Ini adalah absen pertama Anda di HP ini.<br><br>Sistem akan <b>MENGUNCI</b> HP ini khusus untuk akun Anda.</p>', icon: 'warning', showCancelButton: true, confirmButtonText: 'Ya, Kunci & Absen', confirmButtonColor: '#003366'
                }).then((result) => {
                    if (result.isConfirmed) {
                        localStorage.setItem('pusk_sintang_bound_qr', decodedText);
                        prosesKeBackend(decodedText, actionType, onFinish);
                    } else if (onFinish) onFinish();
                });
                return;
            }
            prosesKeBackend(decodedText, actionType, onFinish);
        }

        function drawPagination(containerId, totalPages, currentPg, funcName) {
            let html = '';
            if(totalPages <= 1) { document.getElementById(containerId).innerHTML = ''; return; }

            html += `<button onclick="${funcName}(${currentPg - 1})" class="px-2 py-1 border border-gray-200 rounded text-gray-600 bg-white hover:bg-gray-50 ${currentPg === 1 ? 'opacity-50 pointer-events-none' : ''}"><i class="fa-solid fa-chevron-left"></i></button>`;

            let startP = Math.max(1, currentPg - 2);
            let endP = Math.min(totalPages, currentPg + 2);
            if(startP > 1) html += `<button onclick="${funcName}(1)" class="px-3 py-1 border border-gray-200 rounded text-gray-600 bg-white hover:bg-gray-50">1</button><span class="px-1 text-gray-400">...</span>`;

            for(let i=startP; i<=endP; i++) {
                if(i === currentPg) html += `<button class="px-3 py-1 border border-pusk-blue rounded bg-pusk-blue text-white font-bold">${i}</button>`;
                else html += `<button onclick="${funcName}(${i})" class="px-3 py-1 border border-gray-200 rounded text-gray-600 bg-white hover:bg-gray-50">${i}</button>`;
            }

            if(endP < totalPages) html += `<span class="px-1 text-gray-400">...</span><button onclick="${funcName}(${totalPages})" class="px-3 py-1 border border-gray-200 rounded text-gray-600 bg-white hover:bg-gray-50">${totalPages}</button>`;

            html += `<button onclick="${funcName}(${currentPg + 1})" class="px-2 py-1 border border-gray-200 rounded text-gray-600 bg-white hover:bg-gray-50 ${currentPg === totalPages ? 'opacity-50 pointer-events-none' : ''}"><i class="fa-solid fa-chevron-right"></i></button>`;

            document.getElementById(containerId).innerHTML = html;
        }

        function updateCutiDayCounter() {
            const startEl = document.getElementById('swal-cuti-mulai');
            const endEl = document.getElementById('swal-cuti-selesai');
            const textEl = document.getElementById('cuti-day-counter-text');
            if (!startEl || !endEl || !textEl) return;
            if (!startEl.value || !endEl.value) { textEl.innerText = 'Total cuti: -'; return; }
            const s = new Date(startEl.value + 'T00:00:00');
            const e = new Date(endEl.value + 'T00:00:00');
            if (isNaN(s.getTime()) || isNaN(e.getTime()) || s > e) { textEl.innerText = 'Total cuti: tanggal belum valid'; return; }
            const total = Math.floor((e - s) / 86400000) + 1;
            textEl.innerText = `Total cuti: ${total} hari`;
        }

    function generatePDFDaftarHadir(tgl, namaKeg, list) {
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF({ orientation: "portrait" });
    const pageWidth = doc.internal.pageSize.width;

    // Format Tanggal Profesional
    const monthNames = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
    const dayNames = ["Minggu", "Senin", "Selasa", "Rabu", "Kamis", "Jumat", "Sabtu"];
    let parts = tgl.split('-');
    let d = new Date(parts[0], parseInt(parts[1])-1, parts[2]);
    let tglFormatted = `${dayNames[d.getDay()]}, ${parseInt(parts[2])} ${monthNames[parseInt(parts[1])-1]} ${parts[0]}`;

    // ==================== HEADER ====================
    doc.setFontSize(13);
    doc.setFont("helvetica", "bold");
    doc.text("DAFTAR HADIR KEGIATAN", pageWidth/2, 18, { align: 'center' });

    // Tabel kecil Nama Kegiatan & Tanggal (titik dua sejajar)
    doc.autoTable({
        startY: 28,
        body: [
            ['Nama Kegiatan', ':', namaKeg.toUpperCase()],
            ['Tanggal',       ':', tglFormatted]
        ],
        theme: 'plain',
        styles: { fontSize: 10, cellPadding: 1.5, textColor: [0, 0, 0] },
        columnStyles: {
            0: { cellWidth: 32, fontStyle: 'normal' },
            1: { cellWidth: 5, halign: 'center' },
            2: { fontStyle: 'normal' }
        },
        margin: { left: 14 }
    });

    // ==================== TABEL UTAMA (PEPET + FONT LEBIH KECIL) ====================
    let tableData = [];
    list.forEach((row, index) => {
        let nipText = row.nip ? `\n${row.nip}` : "";
        let golText = row.golongan && row.golongan !== "-" ? `\n${row.golongan}` : "";
        let namaGabung = row.nama + golText + nipText;

        let sig = (index % 2 === 0)
            ? `${index+1}. ......................`
            : `                  ${index+1}. ......................`;

        tableData.push([
            index + 1,
            namaGabung,
            row.jabatan,
            sig
        ]);
    });

    doc.autoTable({
        startY: doc.lastAutoTable.finalY + 4,   // lebih rapat ke atas
        head: [['NO', 'NAMA / PANGKAT / GOL / NIP', 'JABATAN', 'TANDA TANGAN']],
        body: tableData,
        theme: 'grid',
        headStyles: {
            fillColor: [255, 255, 255],
            textColor: [0, 0, 0],
            lineColor: [0, 0, 0],
            lineWidth: 0.1,
            fontStyle: 'bold',
            halign: 'center',
            minCellHeight: 11
        },
        styles: {
            fontSize: 8.5,        // font dikecilkan lagi
            cellPadding: 2.5,     // padding lebih pepet
            valign: 'middle',
            lineColor: [0, 0, 0],
            lineWidth: 0.1,
            textColor: [0, 0, 0]
        },

function populateLokasiFilter() {
    const select = document.getElementById('filterLokasi');
    const lokasiSet = new Set();

    listPegawaiRaw.forEach(p => {
        if (p[5] && p[5].toString().trim() !== '') {
            lokasiSet.add(p[5].toString().trim());
        }
    });

    select.innerHTML = '<option value="SEMUA">Semua Lokasi</option>';

    Array.from(lokasiSet).sort().forEach(lok => {
        let opt = document.createElement('option');
        opt.value = lok;
        opt.textContent = lok;
        select.appendChild(opt);
    });
}

        function gantiHalPegawai(pg) { currentPgPegawai = pg; renderTablePegawai(false); }

function renderTablePegawai(resetPg = false) {
    if(resetPg) currentPgPegawai = 1;

    const searchVal = document.getElementById('searchPegawai').value.toLowerCase().trim();
    const filterLok = document.getElementById('filterLokasi').value;

    filteredPegawai = listPegawaiRaw;

    // Filter Lokasi
    if (filterLok !== 'SEMUA') {
        filteredPegawai = filteredPegawai.filter(p => {
            let lokasi = (p[5] || '').toString().trim();
            return lokasi === filterLok;
        });
    }

    // Filter Search (Nama atau NIP)
    if(searchVal) {
        filteredPegawai = filteredPegawai.filter(p =>
            (p[1] || '').toString().toLowerCase().includes(searchVal) ||
            (p[2] || '').toString().toLowerCase().includes(searchVal)
        );
    }

    let tRows = filteredPegawai.length;
    let tPages = rowsPerPagePegawai === 'ALL' ? 1 : Math.ceil(tRows / rowsPerPagePegawai);
    let sIdx = rowsPerPagePegawai === 'ALL' ? 0 : (currentPgPegawai - 1) * rowsPerPagePegawai;
    let eIdx = rowsPerPagePegawai === 'ALL' ? tRows : sIdx + parseInt(rowsPerPagePegawai);
    let pagedData = filteredPegawai.slice(sIdx, eIdx);

    let html = '';
    if(tRows === 0) {
        html = `<tr><td colspan="4" class="text-center py-10 text-gray-500">Data tidak ditemukan</td></tr>`;
    }
    else {
        pagedData.forEach(p => {
            let golVal = p[6] && p[6] !== "-" ? p[6] : "-";
            let badgeLoc = p[5].includes("Polindes") ? 'bg-purple-50 text-purple-700' : 'bg-blue-50 text-blue-700';

            let eId = p[0] || '';
            let eNama = (p[1] || '').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");
            let eNip = (p[2] || '').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");
            let eJab = (p[3] || '').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");
            let eStat = (p[4] || '').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");
            let eLok = (p[5] || '').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");
            let eGol = (p[6] || '').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");
            let eKunci = (p[7] || 'IYA').toString().replace(/'/g, "\\'").replace(/"/g, "&quot;");

            html += `<tr class="hover:bg-gray-50 border-b border-gray-100">
                <td class="px-4 py-3"><div class="font-bold text-gray-800 text-[12px]">${p[1]}</div><div class="text-[10px] font-mono text-gray-500">${p[2] || '-'}</div></td>
                <td class="px-4 py-3"><div class="text-xs font-bold text-pusk-blue">${golVal}</div><span class="px-2 py-0.5 text-[9px] font-bold uppercase rounded bg-gray-100 text-gray-600">${p[4]}</span></td>
                <td class="px-4 py-3"><div class="text-[11px] text-gray-700 font-bold mb-1">${p[3]}</div><span class="px-2 py-1 text-[9px] font-bold uppercase rounded border ${badgeLoc}">${p[5]}</span></td>
                <td class="px-4 py-3 text-center"><div class="flex justify-center gap-1">
                    <button onclick="bukaModalEdit('${eId}','${eNama}','${eNip}','${eJab}','${eStat}','${eLok}', '${eGol}','${eKunci}')" class="text-white bg-yellow-500 hover:bg-yellow-600 px-2 py-1.5 rounded text-xs"><i class="fa-solid fa-pen"></i></button>
                    <button onclick="hapusPegawai('${eId}')" class="text-white bg-red-500 hover:bg-red-600 px-2 py-1.5 rounded text-xs"><i class="fa-solid fa-trash"></i></button>
                    <button onclick="previewCard('${eId}', '${eNama}', '${eNip}')" class="text-pusk-blue bg-blue-50 border border-blue-200 px-2 py-1.5 rounded text-xs font-bold"><i class="fa-solid fa-id-badge"></i></button>
                </div></td></tr>`;
        });
    }
    document.getElementById('tbPegawai').innerHTML = html;

    document.getElementById('infoPagePegawai').innerText = tRows === 0 ? `0 dari 0` : `${sIdx + 1}-${Math.min(eIdx, tRows)} dari ${tRows}`;
    drawPagination('pagControlsPegawai', tPages, currentPgPegawai, 'gantiHalPegawai');
}

        function bukaModalEdit(id, nama, nip, jabatan, status, lokasi, golongan, kunci) {
            document.getElementById('modalTitlePegawai').innerText = "Edit Data Pegawai";
            document.getElementById('inIdQR').value = id;
            document.getElementById('inNama').value = nama;
            document.getElementById('inNip').value = nip !== '-' ? nip : '';
            document.getElementById('inJabatan').value = jabatan;

            status = status ? status.trim() : "";
            lokasi = lokasi ? lokasi.trim() : "";
            golongan = golongan ? golongan.trim() : "";

            let statEl = document.getElementById('inStatus');
            let statOpts = Array.from(statEl.options).map(o => o.value);
            if(statOpts.includes(status)) statEl.value = status; else statEl.selectedIndex = 0;

            let lokEl = document.getElementById('inLokasi');
            let lokOpts = Array.from(lokEl.options).map(o => o.value);
            if(lokOpts.includes(lokasi)) lokEl.value = lokasi; else lokEl.selectedIndex = 0;

            let golEl = document.getElementById('inGolongan');
            let golOpts = Array.from(golEl.options).map(o => o.value);
            if(golOpts.includes(golongan)) golEl.value = golongan; else golEl.value = "-";
            document.getElementById('inKunci').value = (kunci || "IYA").toUpperCase() === "TIDAK" ? "TIDAK" : "IYA";

            document.getElementById('modalTambahPegawai').classList.remove('hidden');
        }

                Swal.fire('Error', 'Gagal memproses gambar.', 'error');
            });
        }

    const monthNames = ["Januari","Februari","Maret","April","Mei","Juni","Juli","Agustus","September","Oktober","November","Desember"];
    const dayNames = ["Minggu","Senin","Selasa","Rabu","Kamis","Jumat","Sabtu"];

        let statusText = row.keterangan;
        if (row.alasan && row.alasan !== "-" && row.alasan !== "") {
